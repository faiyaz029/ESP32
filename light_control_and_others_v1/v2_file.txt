void loop() {
  if (millis() - lastRPMCheck > 1000) {
    rpm = (pulseCount * 60) / 2; // Assuming 2 pulses per revolution for fan RPM
    pulseCount = 0;
    lastRPMCheck = millis();
  }

  if (millis() - lastWebsiteSend > websiteInterval) {
    sendWebsiteData();
    blinkLED(4);  
    lastWebsiteSend = millis();
  }

  if (millis() - lastRestart > restartInterval) {
    bot.sendMessage(CHAT_ID_1, "⚠️ System restarting as scheduled.", "");
    blinkLED(5);  
    delay(1000);
    ESP.restart();
  }

  if (waveMode && millis() - lastWave > 50) {  
    lightLevel = (int)(127.5 * (sin(millis() / 500.0) + 1.0)); // Sine wave for brightness
    ledcWrite(LIGHT_PWM_CHANNEL, lightLevel);
    lastWave = millis();
  }

  if (WiFi.status() == WL_CONNECTED) {
    int numNewMessages = bot.getUpdates(bot.last_message_received + 1);
    while (numNewMessages) {
      handleNewMessages(numNewMessages);
      numNewMessages = bot.getUpdates(bot.last_message_received + 1);
    }
  }

  server.handleClient();
  delay(50);  
}

void sendStatus() {
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();
  String msg;
  if (!isnan(temp) && !isnan(hum)) {
    msg = " **Status Update** \n";
    msg += " Temp: " + String(temp, 1) + " °C\n";
    msg += "Humidity: " + String(hum, 1) + " %\n";
    msg += "Fan: " + fanMode + " (" + String(fanDuty) + ")\n";
    msg += "RPM: " + String(rpm) + "\n";
    msg += "Light: " + lightMode + " (" + String(lightLevel) + ")";
    if (waveMode) msg += " 🌊";
  } else {
    msg = "❌ DHT22 sensor read error. Please check wiring.";
    Serial.println("DHT22 read error.");
  }
  bot.sendMessage(CHAT_ID_1, msg, "Markdown");
}

void sendWebsiteData() {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("sendWebsiteData: WiFi not connected.");
    return;
  }
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();
  if (isnan(temp) || isnan(hum)) {
    Serial.println("sendWebsiteData: DHT read error, not sending data.");
    return;
  }

  HTTPClient http;
  http.begin(serverUrl);
  http.addHeader("Content-Type", "application/json");

  StaticJsonDocument<256> doc;  
  doc["temperature"] = temp;
  doc["humidity"] = hum;
  // doc["fanMode"] = fanMode;
  // doc["rpm"] = rpm;
  // doc["lightMode"] = lightMode;
  // doc["lightLevel"] = lightLevel;  

  String payload;
  serializeJson(doc, payload);

  Serial.print("Sending data to website: ");
  Serial.println(payload);
  int httpResponseCode = http.POST(payload);

  if (httpResponseCode > 0) {
    Serial.print("HTTP Response code: ");
    Serial.println(httpResponseCode);
  } else {
    Serial.print("Error sending POST: ");
    Serial.println(httpResponseCode);
  }
  http.end();
}

void blinkLED(int times) {
  for (int i = 0; i < times; i++) {
    digitalWrite(LED_PIN, HIGH); delay(150);
    digitalWrite(LED_PIN, LOW); delay(150);
  }
}

void handleSetFan() {
  if (server.hasArg("mode")) {
    String m = server.arg("mode");
    if (m == "slow") setFanSpeed(85, "Slow");  
    else if (m == "medium") setFanSpeed(170, "Medium");
    else if (m == "fast") setFanSpeed(255, "Fast");
    
    server.sendHeader("Location", "/", true);
    server.send(302, "text/plain", "");  
  } else {
    server.send(400, "text/plain", "Bad Request: Missing 'mode' parameter.");
  }
}

void handleSetLight() {
  bool success = false;
  if (server.hasArg("mode")) {
    String m = server.arg("mode");
    waveMode = false;  
    if (m == "dim") { setLightBrightness(50, "Dim"); success = true; }
    else if (m == "bright") { setLightBrightness(255, "Bright"); success = true; }
    else if (m == "off") { setLightBrightness(0, "Off"); success = true; }
  } else if (server.hasArg("value")) {  
    String valStr = server.arg("value");
    int value = valStr.toInt();
    bool isValidNumber = true;
    if (valStr.length() == 0) isValidNumber = false; // Handle empty string
    for (size_t i = 0; i < valStr.length(); i++) {
        if (!isDigit(valStr.charAt(i))) {
            isValidNumber = false;
            break;
        }
    }
    if (isValidNumber && value >= 0 && value <= 255) {
        setLightBrightness(value, "Manual");  
        success = true;
    }
  }

  if (success) {
    server.sendHeader("Location", "/", true);
    server.send(302, "text/plain", "");  
  } else {
    server.send(400, "text/plain", "Bad Request: Missing or invalid 'mode' or 'value' parameter.");
  }
}
