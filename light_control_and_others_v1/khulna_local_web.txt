<script>
function sendCmd(cmd) {
  fetch(`/cmd?do=${encodeURIComponent(cmd)}`) // Encoded cmd just in case
    .then(response => response.text())
    .then(text => {
      console.log(`Sent: ${cmd}, Response: ${text}`);
      if (cmd === 'GET_STATUS') {
        const lines = text.split('\\n'); // Assuming C++ sends literal \n in the string
        lines.forEach(line => {
          const parts = line.split(': ');
          if (parts.length === 2) {
            const key = parts[0].trim();
            const value = parts[1].trim();
            if (key === 'Brightness') {
              document.getElementById('brightnessVal').textContent = value;
            } else if (key === 'Temperature') {
              document.getElementById('tempVal').textContent = value.split(' ')[0];
            } else if (key === 'Humidity') {
              document.getElementById('humidityVal').textContent = value.split(' ')[0];
            }
          }
        });
      }
    })
    .catch(error => console.error('Error sending command:', error));
}

  document.getElementById('lightVal').textContent = numVal;

}

// Helper: send Sony command with repeats
void sendSonyCmd(uint16_t code, uint16_t bits) {
  irsend.sendSony(code, bits, SONY_REPEATS);
  delay(40); // Small delay after sending Sony command
}

// Helper: send Sony input toggle multiple times with delay (kept for potential future use, not used by current HTML)
void sendSonyInputToggleRepeated(int times) {
  for (int i = 0; i < times; i++) {
    irsend.sendSony(SONY_CODE_INPUT_TOGGLE, SONY_BITS_INPUT_TOGGLE, SONY_REPEATS);
    delay(500); // Delay between repeated input toggles
  }
}

void setup() {
  Serial.begin(115200);
  irsend.begin();
  ac.begin(); // Initialize AC object
  ac.on(); // Default AC state to ON, then can be controlled. Or keep it off.
  ac.setFan(kCoolixFanMin); // Default fan
  ac.setMode(kCoolixCool);  // Default mode
  ac.setTemp(24);           // Default temp
  // ac.send(); // Send initial state if needed, or wait for user command

  dht.begin();

  // Setup PWM for light control on channel 0
  ledcAttachPin(LIGHT_PIN, 0); // Attach pin to channel
  ledcSetup(0, 5000, 8);       // channel 0, 5kHz, 8-bit resolution
  ledcWrite(0, brightness);    // Set initial brightness

  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConnected! IP: " + WiFi.localIP().toString());

  if (MDNS.begin("faiyaz")) { // hostname for mDNS
    Serial.println("mDNS responder started at http://faiyaz.local");
  } else {
    Serial.println("Error starting mDNS");
  }

  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
    request->send(200, "text/html", htmlPage);
  });

  server.on("/cmd", HTTP_GET, [](AsyncWebServerRequest *request){
    if (!request->hasParam("do")) {
      request->send(400, "text/plain", "Missing command");
      return;
    }
    String cmd = request->getParam("do")->value();
    cmd.trim();
    Serial.println("Received command: " + cmd);

    bool acCommandSent = false;

    // TV Controls
    if (cmd == "TV_POWER") sendSonyCmd(SONY_CODE_POWER, SONY_BITS_POWER);
    else if (cmd == "TV_VOLUP") sendSonyCmd(SONY_CODE_VOL_UP, SONY_BITS_VOL_UP);
    else if (cmd == "TV_VOLDOWN") sendSonyCmd(SONY_CODE_VOL_DOWN, SONY_BITS_VOL_DOWN);
    else if (cmd == "TV_MUTE") sendSonyCmd(SONY_CODE_MUTE, SONY_BITS_MUTE);
    else if (cmd == "TV_UP") sendSonyCmd(SONY_CODE_UP, SONY_BITS_UP);
    else if (cmd == "TV_DOWN") sendSonyCmd(SONY_CODE_DOWN, SONY_BITS_DOWN);
    else if (cmd == "TV_LEFT") sendSonyCmd(SONY_CODE_LEFT, SONY_BITS_LEFT);
    else if (cmd == "TV_RIGHT") sendSonyCmd(SONY_CODE_RIGHT, SONY_BITS_RIGHT);
    else if (cmd == "TV_OK") sendSonyCmd(SONY_CODE_OK, SONY_BITS_OK);
    else if (cmd == "TV_HOME") sendSonyCmd(SONY_CODE_HOME, SONY_BITS_HOME);
    else if (cmd == "TV_OPTION") sendSonyCmd(SONY_CODE_OPTION, SONY_BITS_OPTION);
    else if (cmd == "TV_RETURN") sendSonyCmd(SONY_CODE_RETURN, SONY_BITS_RETURN);
    else if (cmd == "TV_INPUT") sendSonyCmd(SONY_CODE_INPUT_TOGGLE, SONY_BITS_INPUT_TOGGLE);
    // Removed HDMI1, HDMI2, ScreenShare direct commands as buttons are removed from HTML
    // else if (cmd == "TV_HDMI1") sendSonyInputToggleRepeated(2); // Kept in case of future use
    // else if (cmd == "TV_HDMI2") sendSonyInputToggleRepeated(3); // Kept in case of future use
    // else if (cmd == "TV_SCREENSHARE") sendSonyInputToggleRepeated(4); // Kept in case of future use

    // AC Controls
    else if (cmd == "AC_ON") { ac.on(); acCommandSent = true; }
    else if (cmd == "AC_OFF") { ac.off(); acCommandSent = true; }
    else if (cmd == "AC_MODE_COOL") { ac.setMode(kCoolixCool); acCommandSent = true; }
    else if (cmd == "AC_MODE_HEAT") { ac.setMode(kCoolixHeat); acCommandSent = true; } // Assuming kCoolixHeat exists
    else if (cmd == "AC_MODE_DRY") { ac.setMode(kCoolixDry); acCommandSent = true; }
    else if (cmd.startsWith("AC_TEMP_")) {
      int temp = cmd.substring(8).toInt();
      if (temp >= 16 && temp <= 30) { // Typical AC temp range
        ac.setTemp(temp);
        acCommandSent = true;
      }
    }
    else if (cmd == "AC_FAN_LOW") { ac.setFan(kCoolixFanMin); acCommandSent = true; }
    else if (cmd == "AC_FAN_MED") { ac.setFan(kCoolixFanMed); acCommandSent = true; } // Corrected from AC_FAN_AUTO if it was for Med
    else if (cmd == "AC_FAN_HIGH") { ac.setFan(kCoolixFanMax); acCommandSent = true; }

    // Light Control
    else if (cmd.startsWith("LIGHT_")) {
      int val = cmd.substring(6).toInt();
      val = constrain(val, 0, 255); // Constrain value
      brightness = val;
      ledcWrite(0, brightness); // Apply brightness to LED
    }

    // Sensor Data
    else if (cmd == "GET_STATUS") {
      float t = dht.readTemperature();
      float h = dht.readHumidity();
      String resp = "Brightness: " + String(brightness);
      if (!isnan(t)) {
        resp += "\\nTemperature: " + String(t, 1) + " C"; // Using \n for JS split
      } else {
        resp += "\\nTemperature: N/A";
      }
      if (!isnan(h)) {
        resp += "\\nHumidity: " + String(h, 1) + " %";
      } else {
        resp += "\\nHumidity: N/A";
      }
      request->send(200, "text/plain", resp);
      return; // Important: return here to not send the generic "Sent: cmd"
    }

    // If an AC command was processed, send the IR signal
    if (acCommandSent) {
      ac.send();
      delay(200); // Small delay after sending AC command
    }

    request->send(200, "text/plain", "Sent: " + cmd);
  });

  server.begin();
  Serial.println("HTTP server started");
}
